--- Main Program 3a ---

C Program:

int main() {
int n, key, flag = 0;

```
printf("Enter number of elements: ");
scanf("%d", &n);

printf("Enter %d elements:\n", n);
for (int i = 0; i < n; i++) {
    scanf("%d", &arr[i]);
}

printf("Enter element to search: ");
scanf("%d", &key);

for (int i = 0; i < n; i++) {
    if (arr[i] == key) {
        printf("Element found at position %d\n", i + 1);

        break;
    }

    printf("Element Not Found\n");
}

return 0;

```

}

--- Exercise 3: Find the Largest Element Using Linear Search ---

C Program:
#include <stdio.h>

int main() {
int n;

```
printf("Enter number of elements: ");
scanf("%d", &n);

int arr[n];
printf("Enter %d elements:\n", n);
for (int i = 0; i < n; i++) {
    scanf("%d", &arr[i]);
}

int max = arr[0];
for (int i = 1; i < n; i++) {
    if (arr[i] > max) {
        max = arr[i];
    }
}

printf("Largest element is: %d\n", max);

return 0;

```

}

Output:
Enter number of elements: 5
Enter 5 elements:
12 45 7 89 23
Largest element is: 89

--- Exercise 2: Count Occurrences of a Given String ---

C Program:
#include <stdio.h>
#include <string.h>

int main() {
int n, count = 0;

```
printf("Enter number of strings: ");
scanf("%d", &n);

char arr[n][50];
char key[50];

printf("Enter %d strings:\n", n);
for (int i = 0; i < n; i++) {
    scanf("%s", arr[i]);
}

printf("Enter string to count: ");
scanf("%s", key);

for (int i = 0; i < n; i++) {
    if (strcmp(arr[i], key) == 0) {
        count++;
    }
}

printf("The string '%s' occurred %d time(s).\n", key, count);

return 0;

```

}

Output:
Enter number of strings: 5
Enter 5 strings:
Apple
Mango
Banana
Mango
Mango
Enter string to count: Mango
The string 'Mango' occurred 3 time(s).
