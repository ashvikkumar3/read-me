
PRACTICAL 2a: CHECK BIT AT Nth POSITION

--- Main Program 2a ---

C Program:
#include <stdio.h>

int main() {
int num, n;

```
printf("Enter a number: ");
scanf("%d", &num);
printf("Enter bit position (0-indexed): ");
scanf("%d", &n);

if (num & (1 << n)) {
    printf("Bit at position %d is SET (1)\n", n);
} else {
    printf("Bit at position %d is CLEAR (0)\n", n);
}

return 0;

```

}

Output:
Enter a number: 10
Enter bit position (0-indexed): 1
Bit at position 1 is SET (1)

--- Exercise 1: Toggle a Bit ---

C Program:
#include <stdio.h>

int main() {
int num, n;

```
printf("Enter a number: ");

return 0;

```

}

Enter bit position to toggle (0-indexed): 1

--- Exercise 2: Count Set Bits ---

C Program:
#include <stdio.h>

int main() {
int num, count = 0;

```
printf("Enter a number: ");
scanf("%d", &num);

int temp = num;
while (temp > 0) {
    count += (temp & 1);
    temp >>= 1;
}

printf("Number of set bits in %d is: %d\n", num, count);

return 0;

```

}

Output:
Enter a number: 13
Number of set bits in 13 is: 3

PRACTICAL 2b: MAXIMUM PAIRWISE BITWISE AND

--- Main Program 2b ---

C Program:
#include <stdio.h>

int main() {
int n, i, j;

```
printf("Enter number of elements in array: ");
scanf("%d", &n);

int arr[n];
printf("Enter %d elements:\n", n);
for (i = 0; i < n; i++) {
    scanf("%d", &arr[i]);
}

int maxAND = 0;
for (i = 0; i < n; i++) {
    for (j = i + 1; j < n; j++) {
        int currentAND = arr[i] & arr[j];
        if (currentAND > maxAND) {
            maxAND = currentAND;
        }
    }
}

printf("Maximum Pairwise AND Value: %d\n", maxAND);

return 0;

```

}

Output:
Enter number of elements in array: 4
Enter 4 elements:
12 10 8 13
Maximum Pairwise AND Value: 12

--- Exercise 2: Count Odd Numbers in an Array ---

C Program:
#include <stdio.h>

int main() {
int n, count = 0;

```
printf("Enter size of array: ");
scanf("%d", &n);

int arr[n];
printf("Enter %d elements:\n", n);
for (int i = 0; i < n; i++) {
    scanf("%d", &arr[i]);
    if (arr[i] & 1) {
        count++;
    }
}

printf("Count of odd numbers: %d\n", count);

return 0;

```

}

Output:
Enter size of array: 5
Enter 5 elements:
1 2 3 4 5
Count of odd numbers: 3
